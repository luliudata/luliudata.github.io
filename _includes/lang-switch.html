{%- comment -%} Renders a tiny "EN | 中" pill that links between the page's language versions. From an English URL we just prepend /zh; from a Chinese URL we strip the leading /zh. Pages that don't have a translated sibling will land on the site's 404 — acceptable for now since the language switch on a post implies a translation exists. {%- endcomment -%} {%- assign current_lang = page.lang | default: site.lang | default: "en" -%} {%- if current_lang == "zh" -%} {%- assign other_lang = "en" -%} {%- assign other_url = page.url | remove_first: "/zh" -%} {%- if other_url == "" -%}{%- assign other_url = "/" -%}{%- endif -%} {%- else -%} {%- assign other_lang = "zh" -%} {%- if page.url == "/" -%} {%- assign other_url = "/zh/" -%} {%- else -%} {%- assign other_url = "/zh" | append: page.url -%} {%- endif -%} {%- endif -%}
{{ site.data.i18n[current_lang].lang_short }} | {{ site.data.i18n[current_lang].switch_to }}
